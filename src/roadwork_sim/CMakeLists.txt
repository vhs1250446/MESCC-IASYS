cmake_minimum_required(VERSION 3.5)
project(roadwork_sim)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(roadwork_interfaces REQUIRED)
find_package(roadwork_core REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
add_executable(simulator_node src/simulator_node.cpp)
ament_target_dependencies(simulator_node rclcpp nav_msgs sensor_msgs std_msgs roadwork_interfaces roadwork_core rosgraph_msgs tf2_msgs visualization_msgs)
install(TARGETS simulator_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch rviz DESTINATION share/${PROJECT_NAME})

ament_package()
